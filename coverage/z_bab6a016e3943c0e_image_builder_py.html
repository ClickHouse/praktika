8def _write_file_from_base64(path: str, content: str) -> str: 

9 payload = base64.b64encode(content.encode("utf-8")).decode("ascii") 

10 return f"printf '%s' '{payload}' | base64 -d > {path}"

130def _praktika_controller_component(name: str): 

131 launcher = """#!/usr/bin/env bash 

132set -euo pipefail 

133 

134TOKEN=$(curl -fsS -X PUT "http://169.254.169.254/latest/api/token" -H "X-aws-ec2-metadata-token-ttl-seconds: 60") 

135REGION=${AWS_DEFAULT_REGION:-$(curl -fsS -H "X-aws-ec2-metadata-token: $TOKEN" http://169.254.169.254/latest/meta-data/placement/region)} 

136INSTANCE_ID=${INSTANCE_ID:-$(curl -fsS -H "X-aws-ec2-metadata-token: $TOKEN" http://169.254.169.254/latest/meta-data/instance-id)} 

137PRAKTIKA_CONTROLLER_ROLE=${PRAKTIKA_CONTROLLER_ROLE:-$(curl -fsS -H "X-aws-ec2-metadata-token: $TOKEN" http://169.254.169.254/latest/meta-data/tags/instance/praktika_role || true)} 

138PRAKTIKA_CONTROLLER_QUEUE=${PRAKTIKA_CONTROLLER_QUEUE:-$(curl -fsS -H "X-aws-ec2-metadata-token: $TOKEN" http://169.254.169.254/latest/meta-data/tags/instance/praktika_queue || true)} 

139PRAKTIKA_PROJECT_SLUG=${PRAKTIKA_PROJECT_SLUG:-$(curl -fsS -H "X-aws-ec2-metadata-token: $TOKEN" http://169.254.169.254/latest/meta-data/tags/instance/praktika_project_slug || true)} 

140if [ -z "$PRAKTIKA_CONTROLLER_ROLE" ] || [ -z "$PRAKTIKA_CONTROLLER_QUEUE" ]; then 

141 echo "praktika_role or praktika_queue instance tag is unavailable" >&2 

142 exit 1 

143fi 

144export HOME=/root 

145export AWS_DEFAULT_REGION="$REGION" 

146export INSTANCE_ID="$INSTANCE_ID" 

147export PRAKTIKA_PROJECT_SLUG 

148export PRAKTIKA_CONTROLLER_ROLE 

149export PRAKTIKA_CONTROLLER_QUEUE 

150exec /usr/local/bin/praktika-controller 

151""" 

152 # Both streams are baked into every image. praktika-controller.log always 

153 # has events; praktika-system.log only fills up when the streamer service 

154 # is activated via the praktika_system_logs instance tag (see below), so 

155 # its CloudWatch stream stays empty (and free) on pools that don't opt in. 

156 collect_list: List[Dict[str, str]] = [ 

157 { 

158 "file_path": "/var/log/praktika-controller.log", 

159 "log_group_name": "/${PRAKTIKA_PROJECT_SLUG}/praktika-controller", 

160 "log_stream_name": "{instance_id}", 

161 "timezone": "UTC", 

162 }, 

163 { 

164 # Kernel / systemd / OOM-killer evidence lives in the journal, not 

165 # in the controller's own log, so ship it to a separate stream. 

166 "file_path": "/var/log/praktika-system.log", 

167 "log_group_name": "/${PRAKTIKA_PROJECT_SLUG}/praktika-system", 

168 "log_stream_name": "{instance_id}", 

169 "timezone": "UTC", 

170 }, 

171 ] 

172 cloudwatch_config_json = json.dumps( 

173 {"logs": {"logs_collected": {"files": {"collect_list": collect_list}}}}, 

174 indent=2, 

175 ) 

176 # NOTE: unquoted heredoc so ${PRAKTIKA_PROJECT_SLUG} expands at runtime; 

177 # {instance_id} has no leading $, so the CloudWatch agent resolves it. 

178 cloudwatch_configure = ( 

179 """#!/usr/bin/env bash 

180set -euo pipefail 

181 

182TOKEN=$(curl -fsS -X PUT "http://169.254.169.254/latest/api/token" -H "X-aws-ec2-metadata-token-ttl-seconds: 60") 

183PRAKTIKA_PROJECT_SLUG=${PRAKTIKA_PROJECT_SLUG:-$(curl -fsS -H "X-aws-ec2-metadata-token: $TOKEN" http://169.254.169.254/latest/meta-data/tags/instance/praktika_project_slug || true)} 

184if [ -z "$PRAKTIKA_PROJECT_SLUG" ]; then 

185 echo "praktika_project_slug instance tag is unavailable" >&2 

186 exit 1 

187fi 

188 

189cat > /etc/praktika/amazon-cloudwatch-agent.json <<EOF 

190""" 

191 + cloudwatch_config_json 

192 + """ 

193EOF 

194 

195# The praktika-system-logs streamer (kernel/OOM/systemd-kill evidence) is baked 

196# into every image but stays off unless the pool opts in via the 

197# praktika_system_logs instance tag. Empty stream => no CloudWatch cost when off. 

198PRAKTIKA_SYSTEM_LOGS=$(curl -fsS -H "X-aws-ec2-metadata-token: $TOKEN" http://169.254.169.254/latest/meta-data/tags/instance/praktika_system_logs || true) 

199case "$(printf '%s' "${PRAKTIKA_SYSTEM_LOGS:-}" | tr '[:upper:]' '[:lower:]')" in 

200 1|true|yes|on|enabled) 

201 systemctl enable --now praktika-system-logs || true 

202 ;; 

203 *) 

204 systemctl disable --now praktika-system-logs || true 

205 ;; 

206esac 

207""" 

208 ) 

209 # Follow the journal for the reasons the controller process can die without 

210 # leaving a trace in its own log: kernel OOM killer (_TRANSPORT=kernel), 

211 # systemd manager kill/restart notices (_PID=1), systemd-oomd, and anything 

212 # systemd logs about (or the controller's cgroup emits under) the unit. 

213 system_log_stream = """#!/usr/bin/env bash 

214set -euo pipefail 

215 

216mkdir -p /var/lib/praktika 

217exec journalctl \\ 

218 --output=short-iso \\ 

219 --follow \\ 

220 --cursor-file=/var/lib/praktika/system-log.cursor \\ 

221 _TRANSPORT=kernel + _PID=1 + _COMM=systemd-oomd \\ 

222 + UNIT=praktika-controller.service + _SYSTEMD_UNIT=praktika-controller.service 

223""" 

224 system_logs_unit = """[Unit] 

225Description=Praktika System Log Streamer 

226After=systemd-journald.service 

227Wants=systemd-journald.service 

228 

229[Service] 

230Type=simple 

231ExecStart=/usr/local/bin/praktika-system-log-stream 

232Restart=always 

233RestartSec=5 

234StandardOutput=append:/var/log/praktika-system.log 

235StandardError=append:/var/log/praktika-system.log 

236 

237[Install] 

238WantedBy=multi-user.target 

239""" 

240 unit = """[Unit] 

241Description=Praktika Controller 

242After=network.target docker.service 

243 

244[Service] 

245Type=simple 

246Environment=HOME=/root 

247ExecStart=/usr/local/bin/praktika-controller-start 

248Restart=always 

249RestartSec=5 

250StandardOutput=append:/var/log/praktika-controller.log 

251StandardError=append:/var/log/praktika-controller.log 

252 

253[Install] 

254WantedBy=multi-user.target 

255""" 

256 commands = [ 

257 "mkdir -p /etc/praktika", 

258 "touch /var/log/praktika-controller.log", 

259 "chmod 0644 /var/log/praktika-controller.log", 

260 _write_file_from_base64("/usr/local/bin/praktika-controller-start", launcher), 

261 "chmod 0755 /usr/local/bin/praktika-controller-start", 

262 _write_file_from_base64( 

263 "/usr/local/bin/praktika-configure-cloudwatch-agent", 

264 cloudwatch_configure, 

265 ), 

266 "chmod 0755 /usr/local/bin/praktika-configure-cloudwatch-agent", 

267 _write_file_from_base64( 

268 "/etc/systemd/system/praktika-controller.service", unit 

269 ), 

270 # Always baked; NOT enabled here. Activation is per-pool at boot via the 

271 # praktika_system_logs instance tag, handled by the cloudwatch configure 

272 # script above. 

273 "touch /var/log/praktika-system.log", 

274 "chmod 0644 /var/log/praktika-system.log", 

275 _write_file_from_base64( 

276 "/usr/local/bin/praktika-system-log-stream", system_log_stream 

277 ), 

278 "chmod 0755 /usr/local/bin/praktika-system-log-stream", 

279 _write_file_from_base64( 

280 "/etc/systemd/system/praktika-system-logs.service", 

281 system_logs_unit, 

282 ), 

283 "systemctl daemon-reload || true", 

284 ] 

285 return { 

286 "name": name, 

287 "platform": "Linux", 

288 "description": "Bake the Praktika controller service into the image", 

289 "commands": commands, 

290 }
