27@dataclass 

28class DockerProxy: 

29 """A single-instance DockerHub pull-through cache backed by S3, running 

30 ``zot`` (an OCI-native registry) as one process. 

31 

32 Replaces the two-process ``nginx + registry:2`` proxy: zot serves DockerHub 

33 images at their native ``library/...`` paths (so Docker's ``registry-mirrors`` 

34 works with no image-reference rewrites) and, with ``manifestCheckInterval`` 

35 set, serves cached manifests and blobs from S3 **without re-contacting 

36 DockerHub** — the behavior that previously required an nginx manifest cache 

37 in front of ``registry:2``. 

38 

39 Shape: a single-instance AutoScalingGroup (min=max=desired=1) so a dead node 

40 is replaced automatically, plus the LaunchTemplate, IAM role, and instance 

41 profile it needs. All are created at construction time and registered into 

42 CloudInfrastructure.Config automatically via its ``docker_proxy`` field. 

43 

44 Data path (see docker_proxy_user_data.sh):: 

45 

46 runner --[registry-mirror]--> zot (:5000) --> DockerHub (first pull only) 

47 | 

48 v 

49 S3 (manifests + blobs) 

50 

51 Credentials at rest: none. The DockerHub PAT is read from SSM at boot 

52 (``dockerhub_pat_ssm``) into zot's sync credentials file; S3 is accessed with 

53 the EC2 instance role. The instance self-registers a Route53 A record 

54 (``dns_record`` in ``dns_zone``) at boot and deletes it on shutdown, so 

55 runners reach it at a stable name. 

56 

57 Example:: 

58 

59 docker_proxy = DockerProxy( 

60 instance_type="c7gn.large", 

61 s3_bucket="ch-docker-mirror", 

62 ) 

63 """ 

64 

65 name: str = "dockerhub-proxy" 

66 # Graviton (arm64). Keep a family ending in "g" so the launch-template AMI 

67 # resolver detects arm64 (it keys off family.endswith("g"), which misses the 

68 # "gn"/"gd" network/disk-optimized variants). 

69 instance_type: str = "c7g.large" 

70 vpc_name: str = "" 

71 ami_id: str = "" # AL2023 arm64/x86_64 resolved at deploy time if empty 

72 # zot release to install. Must be >= v2.1.21 (needs manifestCheckInterval). 

73 zot_version: str = "v2.1.21" 

74 # S3 bucket for the mirror storage (manifests + blobs). Reused across 

75 # instance replacements so a fresh node boots warm. 

76 s3_bucket: str = "ch-docker-mirror" 

77 s3_region: str = "us-east-1" 

78 # S3 key prefix zot stores under; isolates zot's layout from any legacy 

79 # registry:2 layout in the same bucket. 

80 s3_rootdirectory: str = "/zot" 

81 upstream_url: str = "https://registry-1.docker.io" 

82 # SSM SecureString holding the DockerHub read-only PAT. Left as-is by project 

83 # namespacing (it is an existing external parameter). 

84 dockerhub_pat_ssm: str = "/ci/docker/robotclickhouse-readonly-token" 

85 dockerhub_username: str = "robotclickhouse" 

86 # How long a cached tag is served without re-checking DockerHub. Bounds 

87 # upstream contact to the first pull of each tag (plus one re-check per tag 

88 # after a process restart, since the last-check time is in-memory). 

89 manifest_check_interval: str = "168h" 

90 listen_port: int = 5000 

91 # Serve zot's web UI (repo/tag browser) on the same port at `/` (the registry 

92 # API stays at `/v2/`). Lightweight: same binary, no CVE/trivy scanning. 

93 enable_ui: bool = False 

94 # Route53 self-registration. The hosted zone must already exist; the record 

95 # is what runners point their registry-mirror at. Both are external DNS 

96 # identities and are NOT project-namespaced. 

97 dns_zone: str = "dockerhub-proxy-zone" 

98 dns_record: str = "dockerhub-proxy.dockerhub-proxy-zone" 

99 security_group_ids: List[str] = field(default_factory=list) 

100 security_group_names: List[str] = field(default_factory=list)

241 def configure_tailscale( 

242 self, 

243 tag, 

244 oauth_client_id_ssm, 

245 oauth_client_secret_ssm, 

246 hostname="", 

247 ): 

248 """Opt into exposing the UI/registry over Tailscale. 

249 

250 Stores the settings in ``ext["tailscale"]`` (keeping the Config lean and 

251 backward-compatible) and re-derives the IAM policy + user_data. At boot the 

252 node mints a tagged ephemeral auth key from the SSM-stored Tailscale OAuth 

253 client, joins the tailnet (``tailscale up --ssh``), and serves the registry 

254 at root over HTTPS via ``tailscale serve``. 

255 

256 ``tag``, ``oauth_client_id_ssm`` and ``oauth_client_secret_ssm`` are 

257 required (there is no safe default for which tag/OAuth client to use). 

258 ``hostname`` is the Tailscale machine name, which becomes the node's 

259 MagicDNS name — the UI is served at ``https://{hostname}.<tailnet>.ts.net/``. 

260 It defaults to the component ``name`` and is a tailnet identity, so it is 

261 NOT project-namespaced. Returns self for chaining.""" 

262 self.ext["tailscale"] = { 

263 "hostname": hostname or self.name, 

264 "tag": tag, 

265 "oauth_client_id_ssm": oauth_client_id_ssm, 

266 "oauth_client_secret_ssm": oauth_client_secret_ssm, 

267 } 

268 self._refresh() 

269 return self

271 def deploy(self): 

272 """Create the private hosted zone (if needed) and authorize inbound to 

273 the proxy's listen port. The IAM role/profile, launch template and ASG 

274 deploy through the standard CloudInfrastructure passes; this method owns 

275 the two things specific to the proxy that those passes don't cover: 

276 

277 1. A private Route53 zone (``dns_zone``) associated with the VPC, so the 

278 instance can self-register ``dns_record`` at boot and runners can 

279 resolve it. praktika has no Route53-zone resource, so the component 

280 owns it here — making the proxy fully deployable by ``--deploy``. 

281 2. An ingress rule for ``listen_port`` on the shared SG (runner -> 

282 proxy), source = the same SG. 

283 

284 This runs before the LaunchTemplate/ASG passes, so on a fresh deploy the 

285 zone exists before the instance boots. Idempotent.""" 

286 from .._utils import aws_client 

287 from ..vpc import VPC 

288 

289 if not self.region: 

290 raise ValueError("DockerProxy.region is not set") 

291 

292 lookup = VPC.Lookup(name=self.vpc_name, region=self.region) 

293 

294 # 1) Ensure the private hosted zone exists and is associated with the VPC. 

295 if self.dns_zone: 

296 self._ensure_hosted_zone(self.region, lookup.vpc_id) 

297 sg_ids = list(self.security_group_ids) 

298 sg_ids.extend(lookup.resolve_security_group_ids(self.security_group_names)) 

299 if not sg_ids: 

300 raise ValueError("DockerProxy has no resolvable security groups") 

301 

302 # 2) Authorize the listen port on the shared SG (runner -> proxy). 

303 ec2 = aws_client("ec2", self.region, "docker-proxy") 

304 for sg_id in sg_ids: 

305 try: 

306 ec2.authorize_security_group_ingress( 

307 GroupId=sg_id, 

308 IpPermissions=[ 

309 { 

310 "IpProtocol": "tcp", 

311 "FromPort": self.listen_port, 

312 "ToPort": self.listen_port, 

313 "UserIdGroupPairs": [{"GroupId": sg_id}], 

314 } 

315 ], 

316 ) 

317 print(f"Authorized tcp/{self.listen_port} on {sg_id} from {sg_id}") 

318 except ec2.exceptions.ClientError as e: 

319 if "InvalidPermission.Duplicate" in str(e): 

320 print(f"Ingress tcp/{self.listen_port} on {sg_id} already present") 

321 else: 

322 raise

324 def _ensure_hosted_zone(self, region: str, vpc_id: str) -> str: 

325 """Ensure a PRIVATE hosted zone named ``dns_zone`` exists and is 

326 associated with ``vpc_id``; create or associate as needed. Returns the 

327 hosted zone id. Idempotent, and never touches zones of other names (the 

328 legacy Terraform ``dockerhub-proxy-zone`` is left alone).""" 

329 from .._utils import aws_client 

330 

331 r53 = aws_client("route53", region, "docker-proxy") 

332 fqdn = self.dns_zone.rstrip(".") + "." 

333 

334 # Private zones with this exact name. 

335 listed = r53.list_hosted_zones_by_name(DNSName=fqdn).get("HostedZones", []) 

336 candidates = [ 

337 z 

338 for z in listed 

339 if z.get("Name") == fqdn and z.get("Config", {}).get("PrivateZone") 

340 ] 

341 for zone in candidates: 

342 detail = r53.get_hosted_zone(Id=zone["Id"]) 

343 associated = {v.get("VPCId") for v in detail.get("VPCs", [])} 

344 if vpc_id in associated: 

345 print(f"Hosted zone {fqdn} already associated with {vpc_id}: {zone['Id']}") 

346 return zone["Id"] 

347 if candidates: 

348 zone_id = candidates[0]["Id"] 

349 r53.associate_vpc_with_hosted_zone( 

350 HostedZoneId=zone_id, 

351 VPC={"VPCRegion": region, "VPCId": vpc_id}, 

352 ) 

353 print(f"Associated hosted zone {fqdn} ({zone_id}) with {vpc_id}") 

354 return zone_id 

355 

356 resp = r53.create_hosted_zone( 

357 Name=fqdn, 

358 CallerReference=f"praktika-docker-proxy-{self.dns_zone}", 

359 HostedZoneConfig={ 

360 "Comment": "praktika DockerHub proxy", 

361 "PrivateZone": True, 

362 }, 

363 VPC={"VPCRegion": region, "VPCId": vpc_id}, 

364 ) 

365 zone_id = resp["HostedZone"]["Id"] 

366 print(f"Created private hosted zone {fqdn}: {zone_id} (associated {vpc_id})") 

367 return zone_id
